cmake_minimum_required(VERSION 3.24)
project(translit_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG v1.3.4 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(minhook)
add_library(translit_hook SHARED hook.cpp)
set_target_properties(translit_hook PROPERTIES OUTPUT_NAME translit_hook_v2)
target_link_libraries(translit_hook PRIVATE minhook d3d11 dxgi)
add_executable(translit_native tools.cpp)
target_link_libraries(translit_native PRIVATE windowsapp advapi32)
add_executable(translit_test_host WIN32 test_host.cpp)
target_link_libraries(translit_test_host PRIVATE d3d11 dxgi d2d1 dwrite)
foreach(t translit_hook translit_native translit_test_host)
 target_compile_definitions(${t} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(${t} PRIVATE /W4 /utf-8 /EHsc)
endforeach()
